async function loadStudy() { try { const res = await fetch('/api/study'); const notes = await res.json(); const grid = document.querySelector('.notes-grid'); grid.innerHTML = notes.map(note => `

${note.title}

${note.description}

${note.pdf_url ? `Download PDF` : ''}
`).join(''); } catch (e) { console.log('Study API failed, using static'); } } loadStudy();
